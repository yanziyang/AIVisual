const scores = keys.map(k =>
  (q[0]*k[0] + q[1]*k[1]) / Math.sqrt(2));
const max = Math.max(...scores);
const exp = scores.map(s => Math.exp(s - max));
const sum = exp.reduce((a, b) => a + b, 0);
const weights = exp.map(e => e / sum);
const output = [0, 1].map(d =>
  values.reduce((s, v, i) => s + weights[i]*v[d], 0));

Subtracting the maximum score makes softmax numerically safer without changing its result.
